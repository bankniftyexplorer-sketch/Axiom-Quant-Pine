// ================================================================
// F2: BIPOWER VARIATION + JUMP DETECTION  [BNS 2004]
//
// BPV  = (π/2) / (n-1) × Σ |r_i| × |r_{i+1}|   close-to-close
// RV   = (1/n) × Σ r_i²                          realized variance
//
// Continuous fraction = min(BPV / RV, 1.0)
//   Applied to YZ: yz_continuous = yz × continuous_fraction
//   This removes jump inflation from the diffusive variance estimate.
//
// Jump flag: jump component exceeds 25% of total variance
//   Returned as float (1.0/0.0) for clean tuple typing
// ================================================================
f_bpv(simple int n) =>
    float rv = 0.0
    for i = 0 to n - 1
        float r = math.log(math.max(close[i], 1e-9) / math.max(close[i + 1], 1e-9))
        rv += r * r
    rv /= n
    float bs = 0.0
    for i = 0 to n - 2
        float a = math.abs(math.log(math.max(close[i],     1e-9) / math.max(close[i + 1], 1e-9)))
        float b = math.abs(math.log(math.max(close[i + 1], 1e-9) / math.max(close[i + 2], 1e-9)))
        bs += a * b
    float bpv = math.pi * 0.5 * bs / math.max(n - 1, 1)
    float cf  = math.min(bpv / math.max(rv, 1e-10), 1.0)
    float jv  = math.max(rv - bpv, 0.0)
    float jfl = rv > 1e-10 and (jv / rv) > 0.25 ? 1.0 : 0.0
    [cf, jv, jfl]

// ================================================================
// F3: R/S STATISTIC  [Hurst 1951]
//
// R = range of cumulative mean-deviation series over N
// S = sample standard deviation of returns over N
// Returns R/S ratio
// Fallback: Anis & Lloyd (1976) E[R/S] for Brownian motion = sqrt(N·π/2)
//           used when S → 0 (flat price — prevents division by zero)
// ================================================================
f_rs(simple int N) =>
    float mu = 0.0
    for i = 0 to N - 1
        mu += math.log(math.max(close[i], 1e-9) / math.max(close[i + 1], 1e-9))
    mu /= N
    float cm = 0.0, hi = -1e18, lo = 1e18, ss = 0.0
    for i = 0 to N - 1
        float r = math.log(math.max(close[i], 1e-9) / math.max(close[i + 1], 1e-9))
        cm += r - mu
        hi  := math.max(hi, cm)
        lo  := math.min(lo, cm)
        ss += (r - mu) * (r - mu)
    float S = math.sqrt(ss / math.max(N - 1, 1))
    S > 1e-10 ? (hi - lo) / S : math.sqrt(float(N) * math.pi * 0.5)

// ================================================================
// F4: HURST EXPONENT  [Hurst 1951; Mandelbrot & Wallis 1969]
//
// H = Δlog(R/S) / Δlog(N) — two-point log-log slope
// Bounded [0.35, 0.65]:
//   Values outside this range indicate insufficient data or regime
//   discontinuity, not genuine extreme persistence. Clamped, not used.
//
// Application (intraday Hurst scaling):
//   σ(T) = σ(1day) × (T / T_session)^(H − 0.5)
//   H = 0.5 → recovers standard √T (Brownian)
//   H < 0.5 → sub-diffusive, narrower intraday bands
//   H > 0.5 → super-diffusive, wider intraday bands
// ================================================================
f_hurst(simple int ns, simple int nl) =>
    float lrs = math.log(math.max(f_rs(nl), 1e-9)) - math.log(math.max(f_rs(ns), 1e-9))
    float lns = math.max(math.log(float(nl)) - math.log(float(ns)), 1e-9)
    math.max(0.35, math.min(0.65, lrs / lns))

// ================================================================
// F5: NORMALIZED SEMIVARIANCE SCALARS  [BNS et al 2010]
//
// RS⁺ = (1/n) Σ r² · 𝟙(r > 0)    upside realized variance
// RS⁻ = (1/n) Σ r² · 𝟙(r < 0)    downside realized variance
//
// Scalar = sqrt(RS_side / (0.5 × RV_total))
//
// NORMALIZATION PROOF:
//   Symmetric market: RS⁺ = RS⁻ = 0.5 × RV
//   → both scalars = sqrt(0.5 / 0.5) = 1.0 ← baseline unchanged
//   Leverage effect: RS⁻ > RS⁺
//   → lower_s > 1.0, upper_s < 1.0 ← asymmetric widening
//   This is the empirically correct behavior for equity indices.
// ================================================================
f_sv(simple int n) =>
    float rp = 0.0, rn = 0.0
    for i = 0 to n - 1
        float r = math.log(math.max(close[i], 1e-9) / math.max(close[i + 1], 1e-9))
        if r > 0.0
            rp += r * r
        else
            rn += r * r
    rp /= n
    rn /= n
    float rt = rp + rn
    [rt > 1e-12 ? math.sqrt(rp / (rt * 0.5)) : 1.0,
     rt > 1e-12 ? math.sqrt(rn / (rt * 0.5)) : 1.0]

// ================================================================
// F6: ROLLING OLS HAR-RV  [Corsi 2009 + Ridge OLS]
//
// MODEL:  RV(t+1) = β₀ + β_d·RV_d + β_w·RV_w + β_m·RV_m
//
// CRITICAL UNIT RULE: all inputs are raw daily variance (not vol,
//   not annualized). Corsi's coefficients only hold in this space.
//   RV_w = 5-bar mean of RV_d   (weekly agent horizon)
//   RV_m = 22-bar mean of RV_d  (monthly agent horizon)
//
// ESTIMATION: β = (X'X + λI)⁻¹ X'Y
//   X[i] = [1, rv_d[i], rv_w[i], rv_m[i]]  for i = 1..ols_n
//   Y[i] = rv_d[i-1]                         one-step-ahead target
//   λ = adaptive ridge = diag_mean × 1e-4    numerical stability
//
// POST-ESTIMATION: β_d, β_w, β_m clamped ≥ 0
//   Economic constraint: variance components cannot negatively
//   forecast tomorrow's variance within the HAR framework.
//
// FALLBACK: Corsi (2009) fixed coefficients when vn < 30
//   β₀=1e-7, β_d=0.36, β_w=0.28, β_m=0.28
//
// NOTE: NaN-guards on matrix.inv() output protect against
//   degenerate matrices on early bars despite ridge regularization.
// ================================================================
f_har(float rv_d, float rv_w, float rv_m, simple int on) =>
    matrix<float> A = matrix.new<float>(4, 4, 0.0)
    matrix<float> C = matrix.new<float>(4, 1, 0.0)
    int vn = 0
    for i = 1 to on
        float x1 = rv_d[i]
        float x2 = rv_w[i]
        float x3 = rv_m[i]
        float y  = rv_d[i - 1]
        if not (na(x1) or na(x2) or na(x3) or na(y))
            matrix.set(A, 0, 0, matrix.get(A, 0, 0) + 1.0      )
            matrix.set(A, 0, 1, matrix.get(A, 0, 1) + x1       )
            matrix.set(A, 0, 2, matrix.get(A, 0, 2) + x2       )
            matrix.set(A, 0, 3, matrix.get(A, 0, 3) + x3       )
            matrix.set(A, 1, 0, matrix.get(A, 1, 0) + x1       )
            matrix.set(A, 1, 1, matrix.get(A, 1, 1) + x1 * x1  )
            matrix.set(A, 1, 2, matrix.get(A, 1, 2) + x1 * x2  )
            matrix.set(A, 1, 3, matrix.get(A, 1, 3) + x1 * x3  )
            matrix.set(A, 2, 0, matrix.get(A, 2, 0) + x2       )
            matrix.set(A, 2, 1, matrix.get(A, 2, 1) + x2 * x1  )
            matrix.set(A, 2, 2, matrix.get(A, 2, 2) + x2 * x2  )
            matrix.set(A, 2, 3, matrix.get(A, 2, 3) + x2 * x3  )
            matrix.set(A, 3, 0, matrix.get(A, 3, 0) + x3       )
            matrix.set(A, 3, 1, matrix.get(A, 3, 1) + x3 * x1  )
            matrix.set(A, 3, 2, matrix.get(A, 3, 2) + x3 * x2  )
            matrix.set(A, 3, 3, matrix.get(A, 3, 3) + x3 * x3  )
            matrix.set(C, 0, 0, matrix.get(C, 0, 0) + y        )
            matrix.set(C, 1, 0, matrix.get(C, 1, 0) + x1 * y   )
            matrix.set(C, 2, 0, matrix.get(C, 2, 0) + x2 * y   )
            matrix.set(C, 3, 0, matrix.get(C, 3, 0) + x3 * y   )
            vn += 1
    float bd = 0.36
    float bw = 0.28
    float bm = 0.28
    float fc = math.max(1e-7 + bd * rv_d + bw * rv_w + bm * rv_m, 1e-10)
    if vn >= 30
        float lam = math.max(
            (matrix.get(A,1,1) + matrix.get(A,2,2) + matrix.get(A,3,3)) / (3.0 * vn) * 1e-4,
            1e-8)
        matrix.set(A, 0, 0, matrix.get(A, 0, 0) + lam)
        matrix.set(A, 1, 1, matrix.get(A, 1, 1) + lam)
        matrix.set(A, 2, 2, matrix.get(A, 2, 2) + lam)
        matrix.set(A, 3, 3, matrix.get(A, 3, 3) + lam)
        matrix<float> B = matrix.mult(matrix.inv(A), C)
        float b0 = matrix.get(B, 0, 0)
        if not na(b0)
            bd := math.max(matrix.get(B, 1, 0), 0.0)
            bw := math.max(matrix.get(B, 2, 0), 0.0)
            bm := math.max(matrix.get(B, 3, 0), 0.0)
            fc  := math.max(b0 + bd * rv_d + bw * rv_w + bm * rv_m, 1e-10)
    [fc, bd, bw, bm]

// ================================================================
// DAILY BUNDLE — evaluated entirely in D context
// All OHLC references, ta.* calls, and [i] indexing
// are on Daily bars when invoked via request.security
//
// _htf: chart-context flag. Computed OUTSIDE request.security so it
// reflects the CHART's timeframe, not the request's D-context.
//   Intraday chart: _htf=false → close = last confirmed D-bar (= yesterday)
//   Daily chart:    _htf=true  → close = today (same TF) → need close[1]
// ================================================================
bool _htf = timeframe.isdaily or timeframe.isweekly or timeframe.ismonthly
f_daily() =>
    // ── 1-bar Rogers-Satchell variance: true 1-day reactivity ────
    // Replaces f_yz(10) to eliminate the collinearity trap.
    // RS = log(H/C)·log(H/O) + log(L/C)·log(L/O)  [always ≥ 0]
    float _lhc_d = math.log(math.max(high,  1e-9) / math.max(close, 1e-9))
    float _llc_d = math.log(math.max(low,   1e-9) / math.max(close, 1e-9))
    float _lho_d = math.log(math.max(high,  1e-9) / math.max(open,  1e-9))
    float _llo_d = math.log(math.max(low,   1e-9) / math.max(open,  1e-9))
    float rv_d   = math.max(_lhc_d * _lho_d + _llc_d * _llo_d, 1e-10)
    // ── BPV: retained for jump diagnostics only (not applied to rv_d) ──
    [cf, jv, jfl]    = f_bpv(i_bpv)
    float rv_w       = ta.sma(rv_d, 5)     // weekly mean  (5 trading days)
    float rv_m       = ta.sma(rv_d, 22)    // monthly mean (22 trading days)
    [hv, bd, bw, bm] = f_har(rv_d, rv_w, rv_m, i_ols)
    float H          = f_hurst(i_rs_s, i_rs_l)
    [us, ls]         = f_sv(i_sv)
    // Vol-of-Vol: σ of annualized HAR vol, de-annualized for price-space use
    float vov        = ta.stdev(math.sqrt(hv * 252.0), 22) / math.sqrt(252.0)
    float anchor     = _htf ? close[1] : close
    [hv, H, us, ls, vov, anchor, jv, jfl, bd, bw, bm]

// Single request.security call — all computation in one D evaluation
[_hv, _H, _us, _ls, _vov, _anch, _jv, _jfl, _bd, _bw, _bm] =
 request.security(syminfo.tickerid, 'D', f_daily(), lookahead=barmerge.lookahead_off)

// ================================================================
// EMPIRICAL DIURNAL NORMALIZATION  [Intraday charts only]
//
// Maintains 7 running means of squared returns, one per NSE hour:
//   Bucket 0=09h, 1=10h, 2=11h, 3=12h, 4=13h, 5=14h, 6=15h
//   All times in Asia/Kolkata (IST = UTC+5:30)
//
// Diurnal factor = bucket_mean / overall_mean_across_buckets
//   > 1.0: current hour historically more volatile than session avg
//   < 1.0: current hour historically calmer than session avg
//   = 1.0: bucket has < 10 observations (neutral, no distortion)
//
// Convergence: ~50 trading days per bucket for stable estimates.
// On daily charts: _intra = false, factor = 1.0 throughout.
//
// WHY NOT HARDCODED: Any fixed multiplier (1.5, 1.3, 0.7) applied
// to NSE time-of-day vol that is NOT derived from your instrument's
// own history is noise dressed as signal. This computes the real thing.
// ================================================================
bool _intra = not (timeframe.isdaily or timeframe.isweekly or timeframe.ismonthly)
var float[] _dsum = array.new_float(7, 0.0)
var int[]   _dcnt = array.new_int(7, 0)

f_bucket() =>
    int h = hour(time, 'Asia/Kolkata')
    h >= 9 and h <= 15 ? h - 9 : -1

// Update running sums (executed every intraday bar)
// Calculates open-to-close return on the first bar to exclude the overnight
// gap, while still capturing the critical opening auction volatility.
if i_dir and _intra and not na(close[1])
    int bk = f_bucket()
    if bk >= 0
        float _dr = session.isfirstbar ? math.log(math.max(close, 1e-9) / math.max(open, 1e-9)) : math.log(math.max(close, 1e-9) / math.max(close[1], 1e-9))
        float sq = _dr * _dr
        array.set(_dsum, bk, array.get(_dsum, bk) + sq)
        array.set(_dcnt, bk, array.get(_dcnt, bk) + 1)

// Compute current bar's diurnal factor
float _df = 1.0
if i_dir and _intra
    int bk = f_bucket()
    if bk >= 0 and array.get(_dcnt, bk) >= 10
        float tot = array.sum(_dsum)
        int   tcn = 0
        for j = 0 to 6
            tcn += array.get(_dcnt, j)
        if tcn > 0
            float ovr    = tot / tcn
            float bkt_mn = array.get(_dsum, bk) / array.get(_dcnt, bk)
            _df := ovr > 1e-12 ? bkt_mn / ovr : 1.0
  
// ================================================================
// INTRADAY SESSION VOLATILITY ENGINE
//
// Tracks realized variance since session open, projects to full
// session length, and blends with the macro HAR forecast.
//
// On intraday charts:
//   anchor = Session VWAP (institutional equilibrium)
//   variance = max(HAR_forecast, projected_session_RV)
//   → bands EXPAND during breakouts, never contract below HAR floor
//
// On daily+ charts:
//   anchor = previous confirmed close (unchanged from original)
//   variance = HAR forecast only (no intraday injection)
//
// Warmup: session RV projection activates after 3 bars to prevent
// wild extrapolation from a single bar's return.
// ================================================================

// ── Session Realized Variance Accumulator ────────────────────────
var float _sess_rv   = 0.0
var int   _sess_bars = 0
if _intra and session.isfirstbar
    _sess_rv   := 0.0
    _sess_bars := 0
if _intra and not na(close[1])
    float _ir = session.isfirstbar ? math.log(math.max(close, 1e-9) / math.max(open, 1e-9)) : math.log(math.max(close, 1e-9) / math.max(close[1], 1e-9))
    _sess_rv  += _ir * _ir
    _sess_bars += 1

// ── Project Session RV to Full Day ───────────────────────────────
// Hurst-aware scaling: (T/t)^(2H) replaces linear (T/t)
// To prevent explosive mathematical singularities when t is near 0,
// elapsed time is clamped to a minimum of 30 minutes for the multiplier.
// Diurnal factor (_df) is now correctly applied to the outer VoV envelope
// rather than corrupting the cumulative integral.
float _tf_min       = math.max(timeframe.in_seconds() / 60.0, 1.0)
float _total_min    = float(i_sess)
float _sess_elapsed = float(_sess_bars) * _tf_min

// Continuous Hurst-aware projection: smoothly forecasts the rest of the day
float _proj_ratio   = _total_min / math.max(_sess_elapsed, 30.0)
float _proj_mult    = math.pow(_proj_ratio, 2.0 * _H)
float _proj_raw     = _sess_rv * _proj_mult

// ── Bayesian Effective Variance Update (RVU) ─────────────────────
// The mathematically perfect formulation for expected daily variance.
// As the session elapses, chronological weight w_hist decreases linearly,
// smoothly transferring trust from the daily forecast (_hv) to the 
// real-time projected variance (_proj_raw).
float _w_hist       = math.max(1.0 - (_sess_elapsed / _total_min), 0.0)
float _w_proj       = 1.0 - _w_hist
float _bayes_hv     = (_hv * _w_hist) + (_proj_raw * _w_proj)

// Generous bounds to prevent math overflows during black swan events,
// while allowing massive dynamic expansion (up to 3.16x width) or 
// tight contraction (down to 0.31x width) based purely on real-time data.
float _eff_hv       = _intra ? math.min(math.max(_bayes_hv, _hv * 0.1), _hv * 10.0) : _hv

// ── Effective Anchor: VWAP on intraday, daily close on D+ ────────
// Warmup: blends VWAP with prev-close over first 60 minutes.
// Suppresses opening-auction noise and smooths post-gap transitions.
float _vwap     = ta.vwap(hlc3)
float _anch_wup = _intra ? math.min(_sess_elapsed / 60.0, 1.0) : 1.0
float _anch_ef  = _intra ? _anch * (1.0 - _anch_wup) + nz(_vwap, _anch) * _anch_wup : _anch

// ================================================================
// BAND CONSTRUCTION
//
// STEP 1: Base expected move (price units)
//   base = anchor × √(effective_variance)
//   Intraday: anchor=VWAP, var=max(HAR, session_RV_projected)
//   Daily+:   anchor=prev_close, var=HAR_forecast
//
// STEP 2: Asymmetric semivariance scaling
//   upper band: ×u_s  (<1.0 in bear regimes due to leverage effect)
//   lower band: ×l_s  (>1.0 in bear regimes — wider downside band)
//
// STEP 3: Vol-of-Vol outer envelope
//   Chebyshev bound: 1.5σ_vov requires no distributional assumption.
//   Compressed envelope = stable regime = bands reliable
//   Wide envelope       = unstable regime = bands indicative only
//
// NOTE: Hurst exponent (H) and Diurnal factor (df) are now active:
//   Hurst: scales session RV projection via (T/n)^(2H) — captures
//     trending vs mean-reverting market structure in band width.
//   Diurnal: dynamically scales the VoV envelope — widens the outer
//     uncertainty bounds during historically volatile hours (e.g. open).
//   Both also displayed in table/CSV for transparency.
// ================================================================
float _base  = _anch_ef * math.sqrt(_eff_hv)
float _em    = _base

float _bhi   = _anch_ef + _em * _us
float _blo   = _anch_ef - _em * _ls
// Apply instantaneous Diurnal Factor to outer uncertainty envelope
float _vdel  = _anch_ef * _vov * 1.5 * math.sqrt(math.max(_df, 0.5))
float _ohi   = _bhi + _vdel
float _olo   = _blo  - _vdel

// ================================================================
// PLOTTING — PREMIUM INSTITUTIONAL VISUALIZATION
//
// Design principles:
//   · Colors are DYNAMIC — adapt to vol-of-vol regime state
//   · Fills simulate gradients via multi-layer phantom plots
//   · Boundary touch/breach markers provide visual confirmation

// ================================================================

// ── VoV Regime Normalization ─────────────────────────────────────
// Maps vol-of-vol to [0, 1] for dynamic color interpolation.
// 0.0 = stable/calm regime   →  muted, confident colors
// 1.0 = chaotic/volatile     →  vivid, warning colors
float _vov_pct = _vov * math.sqrt(252.0) * 100.0
float _vov_n   = math.max(0.0, math.min(_vov_pct / 25.0, 1.0))

// ── Institutional Core Palette ───────────────────────────────────
color c_res_core  = color.rgb(255, 30, 80)    // Razor Red

// ── Inverse Alpha Architecture ───────────────────────────────────
// High VoV = Highly visible outer risk zones (Alpha 70)
