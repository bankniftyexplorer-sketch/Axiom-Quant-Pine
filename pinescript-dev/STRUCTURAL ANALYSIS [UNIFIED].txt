f_dominant_cycle(float src, int min_period = 8, int max_period = 48) =>
    var float period  = 0.0
    var float smooth  = 0.0
    var float detrend = 0.0
    var float i1 = 0.0, var float q1 = 0.0
    var float jI = 0.0, var float jQ = 0.0
    var float i2 = 0.0, var float q2 = 0.0
    var float re = 0.0, var float im = 0.0

    // 1. Noise Elimination (Finite Impulse Response Smoothing)
    smooth := (4.0 * src + 3.0 * nz(src[1]) + 2.0 * nz(src[2]) + nz(src[3])) / 10.0

    // 2. Hilbert Detrending (Amplitude normalization scaling locked to the previous period state)
    float p_prev = nz(period[1], min_period)
    float alpha  = (0.0962 * smooth + 0.5769 * nz(smooth[2]) - 0.5769 * nz(smooth[4]) - 0.0962 * nz(smooth[6])) 
    detrend      := alpha * (0.075 * p_prev + 0.54)

    // 3. Synthesize In-Phase (I) and Quadrature (Q) complex states
    q1 := (0.0962 * detrend + 0.5769 * nz(detrend[2]) - 0.5769 * nz(detrend[4]) - 0.0962 * nz(detrend[6])) * (0.075 * p_prev + 0.54)
    i1 := nz(detrend[3])

    // 4. Force 90-Degree Phase Orthogonality Advance
    jI := (0.0962 * i1 + 0.5769 * nz(i1[2]) - 0.5769 * nz(i1[4]) - 0.0962 * nz(i1[6])) * (0.075 * p_prev + 0.54)
    jQ := (0.0962 * q1 + 0.5769 * nz(q1[2]) - 0.5769 * nz(q1[4]) - 0.0962 * nz(q1[6])) * (0.075 * p_prev + 0.54)

    // 5. Phasor Spatial Addition
    i2 := i1 - jQ
    q2 := q1 + jI

    // 6. Vector Component Stabilization
    i2 := 0.2 * i2 + 0.8 * nz(i2[1])
    q2 := 0.2 * q2 + 0.8 * nz(q2[1])

    // 7. Homodyne Discriminator (Calculates purely the geometric Phase Rate of Change per bar)
    re := i2 * nz(i2[1]) + q2 * nz(q2[1])
    im := i2 * nz(q2[1]) - q2 * nz(i2[1])

    re := 0.2 * re + 0.8 * nz(re[1])
    im := 0.2 * im + 0.8 * nz(im[1])

    // 8. Wavelength Measurement (Bars per Cycle = 2pi Radians / Phase Velocity)
    float calc_period = p_prev
    if re != 0 and im != 0
        calc_period := 2.0 * math.pi / math.atan(im / re)

    // 9. Kinetic Boundary Enforcement & Shock Protection
    // Prevents extreme volatile gaps from breaking the algorithm's phase lock
    calc_period := math.max(calc_period, p_prev * 0.67) 
    calc_period := math.min(calc_period, p_prev * 1.5)
    calc_period := math.max(calc_period, min_period)
    calc_period := math.min(calc_period, max_period)

    // 10. Final Exponential Smoothing of the Temporal Array
    period := 0.2 * calc_period + 0.8 * p_prev

    period

// [OPT-3] nz(V, 0) — consistent with HTF Footprint [F1]
// Inside request.security_lower_tf, nz(volume, 1) is passed — but the function's own
// nz(V, 0) overrides it. This ensures zero volume contributes zero delta.
f_CustomSplitDelta(float O, float H, float L, float C, float V, int dir) =>
    tw         = math.max(H, C[1]) - math.max(O, C)
    bw         = math.min(O, C)    - math.min(L, C[1])
    body       = math.abs(O - C)
    sm         = bw + (C - O) - tw
    cond       = sm >= 0
    total_sq   = (tw * tw) + (bw * bw) + (body * body)
    buy_share  = (bw * bw) + (cond      ? (body * body) : 0)
    sell_share = (tw * tw) + (not cond  ? (body * body) : 0)
    retU       = total_sq > 0 ? (buy_share  / total_sq) : 0.5
    retD       = total_sq > 0 ? (sell_share / total_sq) : 0.5
    dir == 1 ? nz(V, 0) * retU : nz(V, 0) * retD

f_Adaptive_Length_Cum(float src, int length) =>
    int   safe_len = math.max(length, 1)
    float decay    = (safe_len - 1) / safe_len
    var float cum_wave = 0.0
    cum_wave := nz(src) + (nz(cum_wave[1]) * decay)
    cum_wave

    // PASS 1 — SCOUT: accumulate this bar's LTF volume
    // =========================================================================

    for i = 0 to main_sz - 1                           // [OPT-1] cached

    // =========================================================================

    // =========================================================================

        int r_idx = 0

            float main_dd  = array.get(main_ltf_dd,  i)
            float main_vol = array.get(main_ltf_vol, i)

            if not na(main_vol) and not na(ref_vol_scaled)

                // [OPT-2] Each assignment on its own line — no comma chaining
                if mode == 'Main'
                    ref_du_scaled  := 0.0
                    ref_dd_scaled  := 0.0
                    ref_vol_scaled := 0.0

                else if mode == 'Reference'
                    main_du  := 0.0
                    main_dd  := 0.0
                    main_vol := 0.0
                    if special_option_consideration
                        float temp    = ref_du_scaled
                        ref_du_scaled := ref_dd_scaled
                        ref_dd_scaled := temp

                else // Both
                    // [F4] Flip also applies in 'Both' mode — intentional.
                    // PUT + Index: index sell = synthetic PUT buy. Without flip, signals oppose.
                    if special_option_consideration
                        float temp    = ref_du_scaled
