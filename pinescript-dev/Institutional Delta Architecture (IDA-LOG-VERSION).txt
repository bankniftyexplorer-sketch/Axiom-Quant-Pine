f_LastHLbar(series float high_src, series float low_src, series float linked_high ,series float linked_low) =>
    var float candidateHigh = na
    candidateHigh := high_src[1] > high_src[2] and high_src[1] > high_src ? math.max(linked_high[2] , linked_high[1] , linked_high): candidateHigh[1]

    var float candidateLow = na
    candidateLow := low_src[1] < low_src[2] and low_src[1] < low_src ? math.min(linked_low[2] ,linked_low[1] , linked_low) : candidateLow[1]
    
    [candidateHigh, candidateLow]

f_mean_cross(mean_src ,o , c ,h ,l) =>
    var float hh = na
    var float ll = na
    var float por = na // point of revarsal

    kalman = k.f_kalman(mean_src,0.1,0.01)
    condH = c[1] > kalman and c <= kalman or h > kalman and math.pow(o*c , 0.5) < kalman
    condL = c[1] < kalman and c >= kalman or l < kalman and math.pow(o*c , 0.5) > kalman

    por := condH ?  math.max(h,h[1]) : condL ? math.min(l[1] ,l) : por[1]
    por

CandleFix(float O , float H , float L , float C ) =>
    fixnanO = fixnan(O)

    fixnanC = fixnan(C)

    [fixnanO , fixnanH , fixnanL , fixnanC , fixnanV ]


f_4LegRsi(float FO, float FH, float FL, float FC, int length, int smooth) =>
    o = ta.ema(ta.rsi(FO, length), smooth)

    l = ta.ema(ta.rsi(math.min(FL,FC[1]), length), smooth)

    h := math.max(o, h, l, c)
    l := math.min(o, h, l, c)
    
    [o , h ,l ,c]

f_getZone(float v) =>
    int z = -1
    if v >= 0.0 and v < 23.6
        z := 0
    else if v >= 23.6 and v < 38.2
        z := 1
    else if v >= 38.2 and v < 50.0
        z := 2
    else if v >= 50.0 and v < 61.8
        z := 3
    else if v >= 61.8 and v < 78.6
        z := 4
    else if v >= 78.6 and v <= 100.0
        z := 5
    z

f_getBucket(float v, int zi) =>
    float bot  = array.get(ZONE_BOTTOM, zi)
    float top  = array.get(ZONE_TOP,    zi)
    float band = (top - bot) / N_BUCKETS
    int   b    = int(math.floor((v - bot) / band))
    math.min(b, N_BUCKETS - 1)  

f_bucketCenter(int zi, int bj) =>
    float bot  = array.get(ZONE_BOTTOM, zi)
    float top  = array.get(ZONE_TOP,    zi)
    float band = (top - bot) / N_BUCKETS
    bot + band * float(bj) + band * 0.5

f_kalmanUpdate(int zi, float new_obs) =>
    float Q      = 0.005
    float R      = 0.5
    float p_prev = array.get(kalman_p,    zi)
    float p_pred = p_prev + Q
    float gain   = p_pred / (p_pred + R)
    float p_new  = (1.0 - gain) * p_pred

    array.set(kalman_p,    zi, p_new)
    array.set(kalman_gain, zi, gain)           

    float prev_gk = array.get(gravity_k, zi)
    float gk_new  = na(prev_gk) ? new_obs : prev_gk + gain * (new_obs - prev_gk)
    array.set(gravity_k, zi, gk_new)
    gk_new

f_updateGravity(float src, float delta_w) =>
    int zi = f_getZone(src)

    if zi >= 0
        int   bj    = f_getBucket(src, zi)
        int   idx   = zi * N_BUCKETS + bj

        array.set(visit_raw,   idx, array.get(visit_raw,   idx) + 1.0)
        array.set(visit_delta, idx, array.get(visit_delta, idx) + math.abs(delta_w))

        float best_w  = -1.0
        int   best_bj = 0
        for j = 0 to N_BUCKETS - 1
            float w = array.get(visit_delta, zi * N_BUCKETS + j)
            if w > best_w
                best_w  := w
                best_bj := j

        float raw_g = f_bucketCenter(zi, best_bj)
        array.set(gravity_raw, zi, raw_g)
        f_kalmanUpdate(zi, raw_g)

    zi

for i = 0 to 23
    array.set(visit_raw,   i, array.get(visit_raw,   i) * DECAY_FACTOR)
    array.set(visit_delta, i, array.get(visit_delta, i) * DECAY_FACTOR)

// ─── Cumulative Net Delta Engine ─────────────────────────────────────────────
f_sdp_cum_delta() =>
    float _tw       = math.max(high, close[1]) - math.max(open, close)
    float _bw       = math.min(open, close)    - math.min(low, close[1])
    float _body     = math.abs(open - close)
    float _tw_sq    = _tw * _tw
    float _bw_sq    = _bw * _bw
    float _body_sq  = _body * _body
    float _total_sq = _tw_sq + _bw_sq + _body_sq
    bool  _bullish  = (_bw + (close - open) - _tw) >= 0
    float _buy_sh   = _bw_sq + (_bullish     ? _body_sq : 0)
    float _sell_sh  = _tw_sq + (not _bullish ? _body_sq : 0)
    float _buy_pct  = _total_sq > 0 ? _buy_sh  / _total_sq : 0.5
    float _sell_pct = _total_sq > 0 ? _sell_sh / _total_sq : 0.5
    float _net_tick = nz(volume, 0) * (_bullish ? _buy_pct  : - _sell_pct)
    ta.cum(_net_tick)

// ─── Cumulative Log Return: LEVEL series for physics engine input ────────
f_cum_log_ret(float src) =>
    float lr = src > 0 and nz(src[1]) > 0 ? math.log(src / nz(src[1])) : 0.0
    ta.cum(lr)

// ─── 31-Point Structural Score ∈ [-1, +1] ───────────────────────────────
f_structural_score(float p) =>
    if bar_index < 10
        [0.0, 0.0]
    else
        float score = 0.0
        float _eps  = 1e-6
        float _dz   = 1e-8

        d1     = p - p[1]
        d2     = p - 2*p[1] + p[2]
        d3     = p - 3*p[1] + 3*p[2] - p[3]
        d4     = p - 4*p[1] + 6*p[2] - 4*p[3] + p[4]

        d2_e   = p - 2*p[2] + p[4]
        d2_o   = p[1] - 2*p[3] + p[5]
        d3_e   = p - 3*p[2] + 3*p[4] - p[6]
        d3_o   = p[1] - 3*p[3] + 3*p[5] - p[7]

        d1_x   = p - p[1] - p[2] + p[3]
        d2_x   = p - p[1] - 2*p[2] + 2*p[3] + p[4] - p[5]
        convex = 2*p - p[1] - p[2]

        E2_1 = d2 / math.max(math.abs(d1), _eps)
        E4_3 = d4 / math.max(math.abs(d3), _eps)

        score += math.abs(d1)     < _dz ? 0.0 : d1     >= 0 ? 1.0 : -1.0
        score += math.abs(d1_x)   < _dz ? 0.0 : d1_x   >= 0 ? 1.0 : -1.0
        score += math.abs(d2)     < _dz ? 0.0 : d2     >= 0 ? 2.0 : -2.0
        score += math.abs(d2_e)   < _dz ? 0.0 : d2_e   >= 0 ? 2.0 : -2.0
        score += math.abs(d2_o)   < _dz ? 0.0 : d2_o   >= 0 ? 2.0 : -2.0
        score += math.abs(d2_x)   < _dz ? 0.0 : d2_x   >= 0 ? 2.0 : -2.0
        score += math.abs(convex) < _dz ? 0.0 : convex >= 0 ? 2.0 : -2.0
        score += math.abs(d3)     < _dz ? 0.0 : d3     >= 0 ? 3.0 : -3.0
        score += math.abs(d3_e)   < _dz ? 0.0 : d3_e   >= 0 ? 3.0 : -3.0
        score += math.abs(d3_o)   < _dz ? 0.0 : d3_o   >= 0 ? 3.0 : -3.0
        score += math.abs(d4)     < _dz ? 0.0 : d4     >= 0 ? 4.0 : -4.0

        score += E2_1 >= nz(E2_1[1]) ? 1.0 : -1.0
        score += E4_3 >= nz(E4_3[1]) ? 1.0 : -1.0

        score += (d1 > _dz and d2 > _dz and d3 > _dz) ? 1.0 : -1.0
        score += (d1 < -_dz and d2 < -_dz and d3 < -_dz) ? -1.0 : 1.0

        score += (d1 > _dz and d2 > _dz and math.abs(d1) > math.abs(nz(d1[1]))) ? 1.0 : -1.0
        score += (d1 < -_dz and d2 < -_dz and math.abs(d1) > math.abs(nz(d1[1]))) ? -1.0 : 1.0

        [score / 31.0, d1]

// ─── Per-Strike Evaluation ──────────────────────────────────────────────
// Returns: [dominance ∈ [-2,+2], weight, ce_velocity, pe_velocity]
f_eval_strike(float ce_price, float pe_price, float idx_atr) =>
    ce_cum = f_cum_log_ret(ce_price)
    pe_cum = f_cum_log_ret(pe_price)

    [ce_score, ce_vel] = f_structural_score(ce_cum)
    [pe_score, pe_vel] = f_structural_score(pe_cum)

    float dominance = ce_score - pe_score

    float ce_chg = math.abs(nz(ce_price) - nz(ce_price[1]))
    float pe_chg = math.abs(nz(pe_price) - nz(pe_price[1]))
    float weight = (ce_chg + pe_chg) / math.max(idx_atr, 1e-6)

    [dominance, weight, ce_vel, pe_vel]

// ─── Aggregated ComparatorScore ∈ [-1, +1] with Regime Filters ──────────
f_comparator_feature(array<float> arr_dom, array<float> arr_wt, array<float> arr_cv, array<float> arr_pv, float idx_price, float _idx_atr) =>
    float sum_dw    = 0.0
    float sum_w     = 0.0
    int   n_vol_exp = 0
    int size = array.size(arr_dom)
    if size > 0
        for i = 0 to size - 1
            d  = array.get(arr_dom, i)
            w  = array.get(arr_wt,  i)
            cv = array.get(arr_cv, i)
            pv = array.get(arr_pv, i)
            sum_dw += d * w
            sum_w  += w
            if cv > 1e-5 and pv > 1e-5
                n_vol_exp += 1
    float raw     = sum_w > 0 ? sum_dw / sum_w : 0.0
    float clamped = math.max(-1.0, math.min(1.0, raw / 2.0))
    // Index structural confidence
    idx_cum = f_cum_log_ret(idx_price)
    [idx_score, _iv] = f_structural_score(idx_cum)
    float idx_activity = math.abs(idx_score)
    float idx_confidence = math.min(1.0, idx_activity / 0.5)
    // Vol expansion penalty (sigmoid)
    float vol_frac    = size > 0 ? float(n_vol_exp) / float(size) : 0.0
    float vol_penalty = 1.0 / (1.0 + math.exp(10.0 * (vol_frac - 0.2)))
    // Noise floor (ATR-based linear ramp)
    float noise_floor   = 0.005 * size
    float noise_penalty = math.min(1.0, sum_w / math.max(noise_floor, 1e-6))
    // Concentration (HHI)
    float hhi = 0.0
    if size > 0 and sum_w > 0
        for i = 0 to size - 1
            float share = array.get(arr_wt, i) / sum_w
            hhi += share * share
    float hhi_uniform  = size > 0 ? 1.0 / float(size) : 1.0
    float conc_ratio   = (hhi - hhi_uniform) / math.max(1.0 - hhi_uniform, 1e-6)
    float conc_penalty = math.max(0.1, math.min(1.0, conc_ratio * 1.0))
    float penalty = vol_penalty * noise_penalty * conc_penalty * idx_confidence
    clamped * penalty

f_relative_rank(float[] arr, float val) =>
    float min_v = val
    float max_v = val
    for i = 0 to array.size(arr) - 1
        float v = array.get(arr, i)
        if not na(v)
            if na(min_v) or v < min_v
                min_v := v
            if na(max_v) or v > max_v
                max_v := v
    float range_v = max_v - min_v
    range_v > 0 ? (val - min_v) * 100.0 / range_v : 100.0
