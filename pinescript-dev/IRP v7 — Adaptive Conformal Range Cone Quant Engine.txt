f_arr_get_float(float[] src, int idx, float fallback) =>
    idx >= 0 and idx < array.size(src) ? array.get(src, idx) : fallback

f_arr_get_int(int[] src, int idx, int fallback) =>
    idx >= 0 and idx < array.size(src) ? array.get(src, idx) : fallback

f_mat_get(matrix<float> src, int row, int col, float fallback) =>
    int rows = matrix.rows(src)
    int cols = matrix.columns(src)
    row >= 0 and row < rows and col >= 0 and col < cols ? matrix.get(src, row, col) : fallback

f_mat_scalar_mul(matrix<float> src, float k) =>
    int rows = matrix.rows(src)
    int cols = matrix.columns(src)
    matrix<float> out = matrix.new<float>(rows, cols, 0.0)
    for r = 0 to rows - 1
        for c = 0 to cols - 1
            matrix.set(out, r, c, f_mat_get(src, r, c, 0.0) * k)
    out

f_mat_add(matrix<float> a, matrix<float> b) =>
    int rows = matrix.rows(a)
    int cols = matrix.columns(a)
    matrix<float> out = matrix.new<float>(rows, cols, 0.0)
    for r = 0 to rows - 1
        for c = 0 to cols - 1
            matrix.set(out, r, c, f_mat_get(a, r, c, 0.0) + f_mat_get(b, r, c, 0.0))
    out

f_mat_identity(int n) =>
    matrix<float> out = matrix.new<float>(n, n, 0.0)
    for i = 0 to n - 1
        matrix.set(out, i, i, 1.0)
    out

f_cal_alpha(string regime) =>

f_cal_warmup_blend(float preset, float adaptive, int cal_sessions) =>
    if cal_sessions < 20
        preset
    else if cal_sessions < 40
        float w = (cal_sessions - 20) / 20.0
        preset * (1.0 - w) + adaptive * w
    else
        adaptive

f_quantile(float[] src, float q, float fallback) =>
    int n = array.size(src)
    if n == 0
        fallback
    else
        float[] tmp = array.copy(src)
        array.sort(tmp, order.ascending)
        int idx = math.max(0, math.min(n - 1, int(math.ceil(math.max(math.min(q, 1.0), 0.0) * n)) - 1))
        f_arr_get_float(tmp, idx, fallback)

f_weighted_quantile(float[] src, float q, float fallback, float decay) =>
    int n = array.size(src)
    if n == 0
        fallback
    else if n == 1
        f_arr_get_float(src, 0, fallback)
    else
        float[] vals = array.copy(src)
        float[] wts = array.new_float(n, 0.0)
        for i = 0 to n - 1
            array.set(wts, i, math.pow(decay, n - 1 - i))
        for i = 1 to n - 1
            float key_val = f_arr_get_float(vals, i, fallback)
            float key_wt  = f_arr_get_float(wts, i, 0.0)
            int j = i - 1
            while j >= 0 and f_arr_get_float(vals, j, key_val) > key_val
                array.set(vals, j + 1, f_arr_get_float(vals, j, key_val))
                array.set(wts,  j + 1, f_arr_get_float(wts, j, 0.0))
                j -= 1
            array.set(vals, j + 1, key_val)
            array.set(wts,  j + 1, key_wt)
        float total_w = 0.0
        for i = 0 to n - 1
            total_w += f_arr_get_float(wts, i, 0.0)
        float thresh = total_w * math.max(math.min(q, 1.0), 0.0)
        float run_w = 0.0
        float out = f_arr_get_float(vals, n - 1, fallback)
        for i = 0 to n - 1
            run_w += f_arr_get_float(wts, i, 0.0)
            if run_w >= thresh
                out := f_arr_get_float(vals, i, fallback)
                break
        out

// ================================================================
// F2: BIPOWER VARIATION + JUMP DETECTION  [BNS 2004]
//
// BPV  = (π/2) / (n-1) × Σ |r_i| × |r_{i+1}|   close-to-close
// RV   = (1/n) × Σ r_i²                          realized variance
//
// Continuous fraction = min(BPV / RV, 1.0)
// Jump flag: jump component exceeds 25% of total variance
// ================================================================
f_bpv(simple int n) =>
    float rv = 0.0
    for i = 0 to n - 1
        float r = math.log(math.max(close[i], 1e-9) / math.max(close[i + 1], 1e-9))
        rv += r * r
    rv /= n
    float bs = 0.0
    for i = 0 to n - 2
        float a = math.abs(math.log(math.max(close[i],     1e-9) / math.max(close[i + 1], 1e-9)))
        float b = math.abs(math.log(math.max(close[i + 1], 1e-9) / math.max(close[i + 2], 1e-9)))
        bs += a * b
    float bpv = math.pi * 0.5 * bs / math.max(n - 1, 1)
    float cf  = math.min(bpv / math.max(rv, 1e-10), 1.0)
    float jv  = math.max(rv - bpv, 0.0)
    float jfl = rv > 1e-10 and (jv / rv) > 0.25 ? 1.0 : 0.0
    [cf, jv, jfl]

// ================================================================
// F3: R/S STATISTIC  [Hurst 1951]
// ================================================================
f_rs(simple int N) =>
    float mu = 0.0
    for i = 0 to N - 1
        mu += math.log(math.max(close[i], 1e-9) / math.max(close[i + 1], 1e-9))
    mu /= N
    float cm = 0.0, hi = -1e18, lo = 1e18, ss = 0.0
    for i = 0 to N - 1
        float r = math.log(math.max(close[i], 1e-9) / math.max(close[i + 1], 1e-9))
        cm += r - mu
        hi := math.max(hi, cm)
        lo := math.min(lo, cm)
        ss += (r - mu) * (r - mu)
    float S = math.sqrt(ss / math.max(N - 1, 1))
    S > 1e-10 ? (hi - lo) / S : math.sqrt(float(N) * math.pi * 0.5)

// ================================================================
// F4: HURST EXPONENT  [Hurst 1951; Mandelbrot & Wallis 1969]
// ================================================================
f_hurst(simple int ns, simple int nl) =>
    float lrs = math.log(math.max(f_rs(nl), 1e-9)) - math.log(math.max(f_rs(ns), 1e-9))
    float lns = math.max(math.log(float(nl)) - math.log(float(ns)), 1e-9)
    math.max(0.35, math.min(0.65, lrs / lns))

// ================================================================
// F5: NORMALIZED SEMIVARIANCE SCALARS  [BNS et al 2010]
// ================================================================
f_sv(simple int n) =>
    float rp = 0.0, rn = 0.0
    for i = 0 to n - 1
        float r = math.log(math.max(close[i], 1e-9) / math.max(close[i + 1], 1e-9))
        if r > 0.0
            rp += r * r
        else
            rn += r * r
    rp /= n
    rn /= n
    float rt = rp + rn
    [rt > 1e-12 ? math.sqrt(rp / (rt * 0.5)) : 1.0,
     rt > 1e-12 ? math.sqrt(rn / (rt * 0.5)) : 1.0]

// ================================================================
// F6: ROLLING OLS HAR-RV  [Corsi 2009 + Ridge OLS]
// ================================================================
f_har(float rv_d, float rv_w, float rv_m, simple int on) =>
    var float _har_b0_lkg = 1e-7
    var float[] _har_beta_lkg = array.from(0.36, 0.28, 0.28)
    matrix<float> A = matrix.new<float>(4, 4, 0.0)
    matrix<float> C = matrix.new<float>(4, 1, 0.0)
    int vn = 0
    for i = 1 to on
        float x1 = rv_d[i]
        float x2 = rv_w[i]
        float x3 = rv_m[i]
        float y  = rv_d[i - 1]
        if not (na(x1) or na(x2) or na(x3) or na(y))
            matrix.set(A, 0, 0, matrix.get(A, 0, 0) + 1.0)
            matrix.set(A, 0, 1, matrix.get(A, 0, 1) + x1)
            matrix.set(A, 0, 2, matrix.get(A, 0, 2) + x2)
            matrix.set(A, 0, 3, matrix.get(A, 0, 3) + x3)
            matrix.set(A, 1, 0, matrix.get(A, 1, 0) + x1)
            matrix.set(A, 1, 1, matrix.get(A, 1, 1) + x1 * x1)
            matrix.set(A, 1, 2, matrix.get(A, 1, 2) + x1 * x2)
            matrix.set(A, 1, 3, matrix.get(A, 1, 3) + x1 * x3)
            matrix.set(A, 2, 0, matrix.get(A, 2, 0) + x2)
            matrix.set(A, 2, 1, matrix.get(A, 2, 1) + x2 * x1)
            matrix.set(A, 2, 2, matrix.get(A, 2, 2) + x2 * x2)
            matrix.set(A, 2, 3, matrix.get(A, 2, 3) + x2 * x3)
            matrix.set(A, 3, 0, matrix.get(A, 3, 0) + x3)
            matrix.set(A, 3, 1, matrix.get(A, 3, 1) + x3 * x1)
            matrix.set(A, 3, 2, matrix.get(A, 3, 2) + x3 * x2)
            matrix.set(A, 3, 3, matrix.get(A, 3, 3) + x3 * x3)
            matrix.set(C, 0, 0, matrix.get(C, 0, 0) + y)
            matrix.set(C, 1, 0, matrix.get(C, 1, 0) + x1 * y)
            matrix.set(C, 2, 0, matrix.get(C, 2, 0) + x2 * y)
            matrix.set(C, 3, 0, matrix.get(C, 3, 0) + x3 * y)
            vn += 1
    float bd = f_arr_get_float(_har_beta_lkg, 0, 0.36)
    float bw = f_arr_get_float(_har_beta_lkg, 1, 0.28)
    float bm = f_arr_get_float(_har_beta_lkg, 2, 0.28)
    float fc = math.max(_har_b0_lkg + bd * rv_d + bw * rv_w + bm * rv_m, 1e-10)
    float ridge_lambda_used = 1e-8
    float har_fallback_used = 0.0
    if vn >= 30
        float tr = 0.0
        for i = 0 to 3
            tr += f_mat_get(A, i, i, 0.0)
        ridge_lambda_used := math.max(1e-6 * tr / 4.0, 1e-8)
        matrix<float> A_reg = f_mat_add(A, f_mat_scalar_mul(f_mat_identity(4), ridge_lambda_used))
        float det_reg = math.abs(matrix.det(A_reg))
        if na(det_reg) or det_reg < 1e-10
            har_fallback_used := 1.0
        else
            matrix<float> B = matrix.mult(matrix.inv(A_reg), C)
            float b0 = f_mat_get(B, 0, 0, na)
            float bd_raw = f_mat_get(B, 1, 0, na)
            float bw_raw = f_mat_get(B, 2, 0, na)
            float bm_raw = f_mat_get(B, 3, 0, na)
            if not na(b0) and not na(bd_raw) and not na(bw_raw) and not na(bm_raw)
                bd := math.max(bd_raw, 0.0)
                bw := math.max(bw_raw, 0.0)
                bm := math.max(bm_raw, 0.0)
                _har_b0_lkg := b0
                array.set(_har_beta_lkg, 0, bd)
                array.set(_har_beta_lkg, 1, bw)
                array.set(_har_beta_lkg, 2, bm)
                fc := math.max(b0 + bd * rv_d + bw * rv_w + bm * rv_m, 1e-10)
            else
                har_fallback_used := 1.0
    [fc, bd, bw, bm, ridge_lambda_used, har_fallback_used]

bool _htf = timeframe.isdaily or timeframe.isweekly or timeframe.ismonthly
f_daily() =>
    float _lhc_d = math.log(math.max(high,  1e-9) / math.max(close, 1e-9))
    float _llc_d = math.log(math.max(low,   1e-9) / math.max(close, 1e-9))
    float _lho_d = math.log(math.max(high,  1e-9) / math.max(open,  1e-9))
    float _llo_d = math.log(math.max(low,   1e-9) / math.max(open,  1e-9))
    float rv_d   = math.max(_lhc_d * _lho_d + _llc_d * _llo_d, 1e-10)
    [cf, jv, jfl] = f_bpv(i_bpv)
    float rv_w = ta.sma(rv_d, 5)
    float rv_m = ta.sma(rv_d, 22)
    [hv, bd, bw, bm, ridge_lambda_used, har_fallback_used] = f_har(rv_d, rv_w, rv_m, i_ols)
    float H = f_hurst(i_rs_s, i_rs_l)
    [us, ls] = f_sv(i_sv)
    float vov = ta.stdev(math.sqrt(hv * 252.0), 22) / math.sqrt(252.0)
    float vov_med = ta.median(vov, 22)
    float anchor = _htf ? close[1] : close
    float avg_range = ta.sma(high - low, 22)
    [hv, H, us, ls, vov, vov_med, anchor, jv, jfl, bd, bw, bm, rv_m, avg_range, ridge_lambda_used, har_fallback_used]

 request.security(syminfo.tickerid, 'D', f_daily(), lookahead=barmerge.lookahead_off)

f_daily_react() =>
    float _lhc_d = math.log(math.max(high,  1e-9) / math.max(close, 1e-9))
    float _llc_d = math.log(math.max(low,   1e-9) / math.max(close, 1e-9))
    float _lho_d = math.log(math.max(high,  1e-9) / math.max(open,  1e-9))
    float _llo_d = math.log(math.max(low,   1e-9) / math.max(open,  1e-9))
    float rv_d = math.max(_lhc_d * _lho_d + _llc_d * _llo_d, 1e-10)
    float rv_w = ta.sma(rv_d, 5)
    float rv_m = ta.sma(rv_d, 22)
    [hv, _, _, _, _, _] = f_har(rv_d, rv_w, rv_m, i_ols)
    float rv_ref = _htf ? rv_d[1] : rv_d
    float hv_ref = _htf ? hv[2] : hv[1]
    float ratio = not na(rv_ref) and not na(hv_ref) ? rv_ref / math.max(hv_ref, 1e-10) : 1.0
    [rv_ref, hv_ref, ratio]

float _vix_raw = _use_vix ? request.security('NSE:INDIAVIX', 'D', close, lookahead=barmerge.lookahead_off, ignore_invalid_symbol=true) : na
float _vix_daily_var = not na(_vix_raw) ? math.pow(_vix_raw / 100.0, 2) / 252.0 : na
float _iv_rv_ratio = not na(_vix_daily_var) and _hv > 1e-12 ? _vix_daily_var / _hv : 1.0
float _iv_rv_factor = math.pow(_iv_rv_ratio, i_vix_wt)
_iv_rv_factor := math.max(0.5, math.min(2.5, _iv_rv_factor))

float _react_lo = 1.0 / math.max(i_rcap, 1.0)
float _react_clamped = math.min(math.max(nz(_react_ratio_raw, 1.0), _react_lo), i_rcap)
float _react_ratio = 1.0 + (_react_clamped - 1.0) * i_react
float _hv_eff = math.max(_hv * _react_ratio * (_use_vix ? _iv_rv_factor : 1.0), 1e-10)

// ================================================================
// EMPIRICAL DIURNAL NORMALIZATION  [NSE index intraday only]
// ================================================================
bool _intra = not (timeframe.isdaily or timeframe.isweekly or timeframe.ismonthly)
bool _use_nse_diurnal = i_dir and _intra and _is_nse_index
var float[] _dsum = array.new_float(7, 0.0)
var float[] _dcnt = array.new_float(7, 0.0)

f_bucket() =>
    int h = hour(time, 'Asia/Kolkata')
    h >= 9 and h <= 15 ? h - 9 : -1

if _use_nse_diurnal and _intra and session.isfirstbar
    for _di = 0 to 6
        array.set(_dsum, _di, f_arr_get_float(_dsum, _di, 0.0) * 0.997)
        array.set(_dcnt, _di, f_arr_get_float(_dcnt, _di, 0.0) * 0.997)

if _use_nse_diurnal and not na(close[1])
    int bk = f_bucket()
    if bk >= 0 and bk < array.size(_dsum) and bk < array.size(_dcnt)
        float _dr = session.isfirstbar ? math.log(math.max(close, 1e-9) / math.max(open, 1e-9)) : math.log(math.max(close, 1e-9) / math.max(close[1], 1e-9))
        float sq = _dr * _dr
        array.set(_dsum, bk, f_arr_get_float(_dsum, bk, 0.0) + sq)
        array.set(_dcnt, bk, f_arr_get_float(_dcnt, bk, 0.0) + 1.0)

float _df = 1.0
if _use_nse_diurnal
    int bk = f_bucket()
    float _bk_cnt = f_arr_get_float(_dcnt, bk, 0.0)
    if bk >= 0 and bk < array.size(_dsum) and bk < array.size(_dcnt) and _bk_cnt >= 10.0
        float tot = array.sum(_dsum)
        float tcn = 0.0
        for j = 0 to 6
            tcn += f_arr_get_float(_dcnt, j, 0.0)
        if tcn > 0
            float ovr = tot / tcn
            float bkt_mn = f_arr_get_float(_dsum, bk, 0.0) / _bk_cnt
            _df := ovr > 1e-12 ? bkt_mn / ovr : 1.0

// ================================================================
// DUAL-BAND CONFORMAL CALIBRATION STATE
// ================================================================
var float _pred_hi = na
var float _pred_lo = na
var float _pred_anch = na
var float _pred_width = na
var float _inner_hi = na
var float _inner_lo = na
var float _inner_width = na
var float _us_norm_p = 1.0
var float _ls_norm_p = 1.0
var float _base_move_lock = na
var float _raw_up_ref = na
var float _raw_dn_ref = na
var float _cal_up_lock = na
var float _cal_dn_lock = na
var float _cal_inner_up_lock = na
var float _cal_inner_dn_lock = na
var int _cal_sessions_lock = 0
var float _prev_cons = 0.0
var float[] _up_ratio_hist = array.new_float(0)
var float[] _dn_ratio_hist = array.new_float(0)
var float _cal_up_state = na
var float _cal_dn_state = na
var float _cal_inner_up_state = na
var float _cal_inner_dn_state = na
var float _cal_alpha_effective = 0.05

if na(_cal_up_state)
    _cal_up_state := _preset_sigma
if na(_cal_dn_state)
    _cal_dn_state := _preset_sigma
if na(_cal_inner_up_state)
    _cal_inner_up_state := _preset_inner_sigma
if na(_cal_inner_dn_state)
    _cal_inner_dn_state := _preset_inner_sigma

while array.size(_up_ratio_hist) > i_cal_len
    array.shift(_up_ratio_hist)
while array.size(_dn_ratio_hist) > i_cal_len
    array.shift(_dn_ratio_hist)

int _cal_sessions = int(math.min(array.size(_up_ratio_hist), array.size(_dn_ratio_hist)))
float _cal_decay_eff = not na(_vov) and not na(_vov_med) and _vov > _vov_med * 1.15 ? math.max(i_decay - 0.015, 0.95) : i_decay
float _conf_up = f_weighted_quantile(_up_ratio_hist, i_target, _preset_sigma, _cal_decay_eff)
float _conf_dn = f_weighted_quantile(_dn_ratio_hist, i_target, _preset_sigma, _cal_decay_eff)
float _conf_inner_up = f_weighted_quantile(_up_ratio_hist, i_inner_target, _preset_inner_sigma, _cal_decay_eff)
float _conf_inner_dn = f_weighted_quantile(_dn_ratio_hist, i_inner_target, _preset_inner_sigma, _cal_decay_eff)
[_cal_lo_clamp, _cal_hi_clamp] = f_cal_clamps(_symbol_key)
float _adaptive_cal_up = math.max(_cal_lo_clamp, math.min(_cal_hi_clamp, nz(_cal_up_state, _preset_sigma)))
float _adaptive_cal_dn = math.max(_cal_lo_clamp, math.min(_cal_hi_clamp, nz(_cal_dn_state, _preset_sigma)))
float _adaptive_inner_up = math.max(_cal_lo_clamp, math.min(_cal_hi_clamp, nz(_cal_inner_up_state, _preset_inner_sigma)))
float _adaptive_inner_dn = math.max(_cal_lo_clamp, math.min(_cal_hi_clamp, nz(_cal_inner_dn_state, _preset_inner_sigma)))
float _next_cal_up = i_adapt_cal ? f_cal_warmup_blend(_preset_sigma, _adaptive_cal_up, _cal_sessions) : _preset_sigma
float _next_cal_dn = i_adapt_cal ? f_cal_warmup_blend(_preset_sigma, _adaptive_cal_dn, _cal_sessions) : _preset_sigma
float _next_inner_up = i_adapt_cal ? f_cal_warmup_blend(_preset_inner_sigma, _adaptive_inner_up, _cal_sessions) : _preset_inner_sigma
float _next_inner_dn = i_adapt_cal ? f_cal_warmup_blend(_preset_inner_sigma, _adaptive_inner_dn, _cal_sessions) : _preset_inner_sigma
float _eff_sigma = i_sigma

float _base_move_live = na
float _raw_up_live = na
float _raw_dn_live = na

// ================================================================
// PREDICTED RANGE — LOCKED AT SESSION OPEN
// ================================================================
if _intra and session.isfirstbar
    float _gap_pct = _anch > 0 ? math.abs(open - _anch) / _anch * 100.0 : 0.0
    _pred_anch := _gap_pct > 0.3 ? open : _anch
    float _var_ratio = _rv_m > 1e-12 ? math.sqrt(math.max(_hv_eff, 1e-10) / _rv_m) : 1.0
    float _us_norm = _us / math.max((_us + _ls) * 0.5, 1e-6)
    float _ls_norm = _ls / math.max((_us + _ls) * 0.5, 1e-6)
    _us_norm_p := _us_norm
    _ls_norm_p := _ls_norm
    _base_move_lock := nz(_avg_range, _pred_anch * math.sqrt(math.max(_hv_eff, 1e-10)) * 2.0) * 0.5 * _var_ratio * _eff_sigma
    _cal_up_lock := _next_cal_up
    _cal_dn_lock := _next_cal_dn
    _cal_inner_up_lock := _next_inner_up
    _cal_inner_dn_lock := _next_inner_dn
    _cal_sessions_lock := _cal_sessions
    _raw_up_ref := _base_move_lock * _us_norm
    _raw_dn_ref := _base_move_lock * _ls_norm
    _pred_hi := _pred_anch + _raw_up_ref * _cal_up_lock
    _pred_lo := _pred_anch - _raw_dn_ref * _cal_dn_lock
    _pred_width := _pred_hi - _pred_lo
    _inner_hi := _pred_anch + _raw_up_ref * _cal_inner_up_lock
    _inner_lo := _pred_anch - _raw_dn_ref * _cal_inner_dn_lock
    _inner_width := _inner_hi - _inner_lo
    _prev_cons := 0.0

if not _intra
    _pred_anch := _anch
    float _var_ratio = _rv_m > 1e-12 ? math.sqrt(math.max(_hv_eff, 1e-10) / _rv_m) : 1.0
    float _us_norm = _us / math.max((_us + _ls) * 0.5, 1e-6)
    float _ls_norm = _ls / math.max((_us + _ls) * 0.5, 1e-6)
    _base_move_live := nz(_avg_range, _pred_anch * math.sqrt(math.max(_hv_eff, 1e-10)) * 2.0) * 0.5 * _var_ratio * _eff_sigma
    _raw_up_live := _base_move_live * _us_norm
    _raw_dn_live := _base_move_live * _ls_norm
    _pred_hi := _pred_anch + _raw_up_live * _next_cal_up
    _pred_lo := _pred_anch - _raw_dn_live * _next_cal_dn
    _pred_width := _pred_hi - _pred_lo
    _inner_hi := _pred_anch + _raw_up_live * _next_inner_up
    _inner_lo := _pred_anch - _raw_dn_live * _next_inner_dn
    _inner_width := _inner_hi - _inner_lo

float _cal_up = _intra ? nz(_cal_up_lock, _next_cal_up) : _next_cal_up
float _cal_dn = _intra ? nz(_cal_dn_lock, _next_cal_dn) : _next_cal_dn
float _cal_inner_up = _intra ? nz(_cal_inner_up_lock, _next_inner_up) : _next_inner_up
float _cal_inner_dn = _intra ? nz(_cal_inner_dn_lock, _next_inner_dn) : _next_inner_dn
int _cal_sessions_used = _intra ? _cal_sessions_lock : _cal_sessions
float _base_move = _intra ? _base_move_lock : _base_move_live
float _raw_up_half = _intra ? _raw_up_ref : _raw_up_live
float _raw_dn_half = _intra ? _raw_dn_ref : _raw_dn_live

// ================================================================
// CONSUMPTION TRACKER
// ================================================================
var float _sess_hi = na
var float _sess_lo = na

if _intra and session.isfirstbar
    _sess_hi := high
    _sess_lo := low
else if _intra
    _sess_hi := math.max(nz(_sess_hi, high), high)
    _sess_lo := math.min(nz(_sess_lo, low), low)

float _consumed_pct = _intra and _pred_width > 0 ? (_sess_hi - _sess_lo) / _pred_width * 100.0 : na

// ================================================================
// SESSION TIMING + OPENING RANGE REFINEMENT
// ================================================================
float _tf_min = math.max(timeframe.in_seconds() / 60.0, 1.0)
float _total_min = float(i_sess)

var int _sess_bars = 0
var float _opening_range_pct = na
var float _or_adj = 1.0
if _intra and session.isfirstbar
    _sess_bars := 1
    _opening_range_pct := na
    _or_adj := 1.0
else if _intra
    _sess_bars += 1

float _sess_elapsed = float(_sess_bars) * _tf_min
float _sess_fraction = math.min(_sess_elapsed / _total_min, 1.0)
bool _past_third = _sess_fraction >= 0.33

float _or_threshold_min = 30.0
if _intra and _sess_elapsed >= _or_threshold_min and _sess_elapsed < _or_threshold_min + _tf_min * 2
    _opening_range_pct := nz(_consumed_pct, 0.0)
    if _opening_range_pct > 35
        _or_adj := 1.20
    else if _opening_range_pct > 25
        _or_adj := 1.10
    else if _opening_range_pct < 10
        _or_adj := 0.85
    else
        _or_adj := 1.0

// ================================================================
// DYNAMIC INTRADAY RANGE CONE
// ================================================================
float _rem_var_frac = _intra ? math.max((_total_min - _sess_elapsed) / _total_min, 0.02) : 1.0

if _use_nse_diurnal
    float _tot_var = array.sum(_dsum)
    if _tot_var > 1e-12
        float _cum_var = 0.0
        float _prev_min = 0.0
        float _prev_cv = 0.0
        bool _found = false
        var float[] _b_mins = array.from(45.0, 105.0, 165.0, 225.0, 285.0, 345.0, 375.0)
        for i = 0 to 6
            float _curr_min = f_arr_get_float(_b_mins, i, _prev_min)
            float _bucket_var = f_arr_get_float(_dsum, i, 0.0) / _tot_var
            float _curr_cv = _prev_cv + _bucket_var
            if _sess_elapsed <= _curr_min and not _found
                float _frac = (_sess_elapsed - _prev_min) / math.max(_curr_min - _prev_min, 1.0)
                _cum_var := _prev_cv + _frac * (_curr_cv - _prev_cv)
                _found := true
            _prev_min := _curr_min
            _prev_cv := _curr_cv
        if _found
            _rem_var_frac := math.max(1.0 - _cum_var, 0.02)

float _hurst_exp = math.max(_H, 0.1)
float _remaining_up = _intra and not na(_base_move) ? _base_move * _cal_up * math.pow(_rem_var_frac, _hurst_exp) * _or_adj : na
float _remaining_dn = _intra and not na(_base_move) ? _base_move * _cal_dn * math.pow(_rem_var_frac, _hurst_exp) * _or_adj : na
float _remaining_up_pts = _intra and not na(_remaining_up) ? _remaining_up * _us_norm_p : na
float _remaining_dn_pts = _intra and not na(_remaining_dn) ? _remaining_dn * _ls_norm_p : na
float _cone_hi = _intra and not na(_remaining_up_pts) ? close + _remaining_up_pts : na
float _cone_lo = _intra and not na(_remaining_dn_pts) ? close - _remaining_dn_pts : na
float _remaining_pts = _intra and not na(_remaining_up_pts) and not na(_remaining_dn_pts) ? _remaining_up_pts + _remaining_dn_pts : na

// CONSUMPTION VELOCITY
// ================================================================
float _cons_velocity = _intra and not na(_consumed_pct) ? _consumed_pct - nz(_prev_cons, 0.0) : 0.0
if _intra and not na(_consumed_pct)
    _prev_cons := _consumed_pct

// ================================================================
// MOMENTUM ENGINE
// ================================================================
float _mom_5 = not na(close[5]) and close[5] > 0 ? (close - close[5]) / close[5] * 10000.0 : 0.0
float _mom_15 = not na(close[15]) and close[15] > 0 ? (close - close[15]) / close[15] * 10000.0 : 0.0
float _accel = not na(_mom_5[5]) ? _mom_5 - nz(_mom_5[5], 0.0) : 0.0
bool _momentum_bullish = _mom_5 > 5.0 and _accel > 0.0
bool _momentum_bearish = _mom_5 < -5.0 and _accel < 0.0
bool _momentum_confirmed = _momentum_bullish or _momentum_bearish

// ================================================================
// DIRECTIONAL LEAN SCORE
// ================================================================
float _dir_lean = 0.0
float _sv_total = math.max(_us_norm_p + _ls_norm_p, 1e-6)
_dir_lean += (_us_norm_p - _ls_norm_p) / _sv_total * 30.0
float _anch_dist = not na(_pred_anch) and _pred_anch > 0 ? (close - _pred_anch) / _pred_anch * 100.0 : 0.0
_dir_lean += math.max(-25.0, math.min(25.0, _anch_dist * 50.0))
float _up_consumed_pct = not na(_sess_hi) and not na(_pred_anch) and not na(_raw_up_half) and _raw_up_half > 0 ? math.max(_sess_hi - _pred_anch, 0.0) / math.max(_raw_up_half * _cal_up, 1e-9) * 100.0 : 0.0
float _dn_consumed_pct = not na(_sess_lo) and not na(_pred_anch) and not na(_raw_dn_half) and _raw_dn_half > 0 ? math.max(_pred_anch - _sess_lo, 0.0) / math.max(_raw_dn_half * _cal_dn, 1e-9) * 100.0 : 0.0
_dir_lean += math.max(-20.0, math.min(20.0, (_up_consumed_pct - _dn_consumed_pct) * 0.20))
_dir_lean += math.max(-25.0, math.min(25.0, _mom_5 * 0.5))
_dir_lean := math.max(-100.0, math.min(100.0, _dir_lean))
string _dir_label = _dir_lean > 15.0 ? 'BULL' : _dir_lean < -15.0 ? 'BEAR' : 'FLAT'
bool _dir_clear = math.abs(_dir_lean) > 15.0

// ================================================================
// BREAKOUT SCORE
// ================================================================
float _dist_inner_up = not na(_inner_hi) and _inner_width > 0 ? (_inner_hi - close) / _inner_width : na
float _dist_inner_dn = not na(_inner_lo) and _inner_width > 0 ? (close - _inner_lo) / _inner_width : na
float _nearest_inner = not na(_dist_inner_up) and not na(_dist_inner_dn) ? math.min(_dist_inner_up, _dist_inner_dn) : na
float _dist_outer_up = not na(_pred_hi) and _pred_width > 0 ? (_pred_hi - close) / _pred_width : na
float _dist_outer_dn = not na(_pred_lo) and _pred_width > 0 ? (close - _pred_lo) / _pred_width : na
float _nearest_outer = not na(_dist_outer_up) and not na(_dist_outer_dn) ? math.min(_dist_outer_up, _dist_outer_dn) : na
bool _beyond_inner = not na(_inner_hi) and not na(_inner_lo) and (close > _inner_hi or close < _inner_lo)
bool _beyond_outer = not na(_pred_hi) and not na(_pred_lo) and (close > _pred_hi or close < _pred_lo)

float _bk_base = 25.0
float _prox_adj = 0.0
if not na(_nearest_inner)
    if _nearest_inner < 0.05
        _prox_adj := 18.0
    else if _nearest_inner < 0.10
        _prox_adj := 10.0
    else if _nearest_inner < 0.20
        _prox_adj := 4.0
    else if _nearest_inner > 0.40
        _prox_adj := -8.0
float _beyond_adj = _beyond_inner ? 15.0 : 0.0
float _beyond_outer_adj = _beyond_outer ? 10.0 : 0.0
float _mom_adj = math.abs(_mom_5) > 20 ? 10.0 : math.abs(_mom_5) > 10 ? 6.0 : math.abs(_mom_5) > 5 ? 3.0 : -3.0
float _accel_adj = _accel > 2.0 ? 5.0 : _accel < -2.0 ? -4.0 : 0.0
float _hurst_bk = _H > 0.56 ? 8.0 : _H < 0.44 ? -6.0 : 0.0
float _time_bk = _sess_fraction < 0.25 ? 6.0 : _sess_fraction < 0.50 ? 2.0 : _sess_fraction > 0.75 ? -10.0 : -3.0
float _vel_bk = _cons_velocity > 12 ? 6.0 : _cons_velocity > 6 ? 3.0 : _cons_velocity < 2 and _sess_fraction > 0.4 ? -6.0 : 0.0
float _jump_bk = _jfl > 0.5 ? 6.0 : 0.0
float _vov_bk = not na(_vov) and not na(_vov_med) and _vov > _vov_med * 1.15 ? 4.0 : 0.0
float _breakout_raw = _bk_base + _prox_adj + _beyond_adj + _beyond_outer_adj + _mom_adj + _accel_adj + _hurst_bk + _time_bk + _vel_bk + _jump_bk + _vov_bk
float _breakout_score = _intra ? math.max(0.0, math.min(100.0, _breakout_raw)) : 0.0

// ================================================================
// ENTRY QUALITY GRADE
// ================================================================
float _time_quality = _sess_fraction < 0.25 ? 80.0 : _sess_fraction < 0.50 ? 60.0 : _sess_fraction < 0.70 ? 30.0 : 10.0
float _mom_quality = math.abs(_mom_5) > 15 ? 70.0 : math.abs(_mom_5) > 8 ? 45.0 : 15.0
float _entry_raw = _breakout_score * 0.40 + math.abs(_dir_lean) * 0.30 + _time_quality * 0.20 + _mom_quality * 0.10
string _entry_grade = _entry_raw >= 70.0 ? 'A' : _entry_raw >= 55.0 ? 'B' : _entry_raw >= 40.0 ? 'C' : _entry_raw >= 25.0 ? 'D' : 'F'
color _entry_col = _entry_grade == 'A' ? color.rgb(0, 220, 160) : _entry_grade == 'B' ? color.rgb(0, 180, 220) : _entry_grade == 'C' ? color.rgb(255, 170, 0) : color.rgb(255, 50, 80)

// ================================================================
// MODEL CONFIDENCE
// ================================================================
bool _ols_active = math.abs(_bd - 0.36) > 1e-9 or math.abs(_bw - 0.28) > 1e-9 or math.abs(_bm - 0.28) > 1e-9
bool _hurst_boundary = math.abs(_H - 0.35) < 1e-9 or math.abs(_H - 0.65) < 1e-9
float _cal_div = math.max(_cal_up / math.max(_cal_dn, 1e-9), _cal_dn / math.max(_cal_up, 1e-9))
bool _bars_ready = not na(_hv) and not na(_H) and not na(_avg_range) and not na(_rv_m)
float _confidence = 100.0
if _cal_sessions_used < 20
    _confidence -= 20.0
if not _ols_active
    _confidence -= 10.0
if not _recognized_symbol
    _confidence -= 8.0
if _is_nse_index and i_vix and na(_vix_raw)
    _confidence -= 8.0
if not _intra
    _confidence -= 10.0
if _hurst_boundary
    _confidence -= 10.0
if _cal_div > 1.8
    _confidence -= 8.0
if not _bars_ready
    _confidence -= 10.0
float _model_confidence = math.max(0.0, math.min(100.0, _confidence))

// ================================================================
// BUYER REGIME ENGINE
// ================================================================
bool _is_breakout = _intra and _beyond_inner
bool _is_loading = _intra and not _is_breakout and ((not na(_nearest_inner) and _nearest_inner < 0.10) or _breakout_score > 55.0)
bool _is_riding = _intra and not _is_breakout and not _is_loading and (nz(_consumed_pct, 0.0) > 55 or _cons_velocity > 10 or (_H > 0.55 and nz(_consumed_pct, 0.0) > 45))
bool _is_dead = _intra and not _is_breakout and not _is_loading and not _is_riding and _past_third and nz(_consumed_pct, 0.0) < 30 and _cons_velocity < 2 and _breakout_score < 20

string _raw_regime = 'WATCH'
if not _intra
    _raw_regime := 'DAILY'
else if _is_breakout
    _raw_regime := 'BREAKOUT'
else if _is_loading
    _raw_regime := 'LOADING'
else if _sess_fraction < 0.33
    _raw_regime := 'BUILDING'
else if _is_riding
    _raw_regime := 'RIDING'
else if _is_dead
    _raw_regime := 'DEAD'

var string _regime_prev = 'WATCH'
var int _regime_dwell_bars = 0
var int _regime_downgrade_wait = 0
string _regime = _raw_regime

if not _intra
    _regime := 'DAILY'
    _regime_prev := _regime
    _regime_dwell_bars := 0
    _regime_downgrade_wait := 0
else if session.isfirstbar
    _regime := _raw_regime
    _regime_prev := _regime
    _regime_dwell_bars := 0
    _regime_downgrade_wait := 0
else
    int _severity_raw = f_regime_severity(_raw_regime)
    int _severity_prev = f_regime_severity(_regime_prev)
    _regime := _regime_prev
    if _severity_raw > _severity_prev
        _regime := _raw_regime
        _regime_prev := _regime
        _regime_dwell_bars := 0
        _regime_downgrade_wait := 0
    else if _severity_raw < _severity_prev
        _regime_downgrade_wait += 1
        _regime_dwell_bars += 1
        if _regime_downgrade_wait >= 3
            _regime := _raw_regime
            _regime_prev := _regime
            _regime_dwell_bars := 0
            _regime_downgrade_wait := 0
    else
        _regime_downgrade_wait := 0
        if _raw_regime != _regime_prev
            _regime := _raw_regime
            _regime_prev := _regime
            _regime_dwell_bars := 0
        else
            _regime := _regime_prev

    array.push(_dn_ratio_hist, _dn_ratio)

    while array.size(_dn_ratio_hist) > i_cal_len

        float _raw_q_dn = f_weighted_quantile(_dn_ratio_hist, i_target, _preset_sigma, _cal_decay_eff)

        float _step_dn = (_raw_q_dn - _target_dn) * _cal_alpha_effective

        _step_dn := math.max(math.min(_step_dn, 0.05), -0.05)

        _cal_dn_state := math.max(_cal_lo_clamp, math.min(_cal_hi_clamp, _target_dn + _step_dn))

        _cal_inner_dn_state := math.max(_cal_lo_clamp, math.min(_cal_hi_clamp, _target_inner_dn + _step_inner_dn))
