f_LastHLbar(series float high_src, series float low_src, series float linked_high ,series float linked_low) =>
    var float candidateHigh = na
    candidateHigh := high_src[1] > high_src[2] and high_src[1] > high_src ? math.max(linked_high[2] , linked_high[1] , linked_high): candidateHigh[1]

    var float candidateLow = na
    candidateLow := low_src[1] < low_src[2] and low_src[1] < low_src ? math.min(linked_low[2] ,linked_low[1] , linked_low) : candidateLow[1]
    
    [candidateHigh, candidateLow]

f_mean_cross(mean_src ,o , c ,h ,l) =>

    kalman = k.f_kalman(mean_src,0.1,0.01)
    condH = c[1] > kalman and c <= kalman or h > kalman and math.pow(o*c , 0.5) < kalman
    condL = c[1] < kalman and c >= kalman or l < kalman and math.pow(o*c , 0.5) > kalman

    por

f_getZone(float v) =>
    int z = -1
    if v >= 0.0   and v <  16.666 
        z := 0
    else if v >= 16.666  and v <  33.333  
        z := 1
    else if v >= 33.333  and v <  49.999  
        z := 2
    else if v >= 49.999  and v <  66.666  
        z := 3
    else if v >= 66.666  and v <  83.333  
        z := 4
    else if v >= 83.333  and v <= 99.999 
        z := 5
    z

f_getBucket(float v, int zi) =>
    float bot  = array.get(ZONE_BOTTOM, zi)
    float top  = array.get(ZONE_TOP,    zi)
    float band = (top - bot) / N_BUCKETS
    int   b    = int(math.floor((v - bot) / band))
    math.min(b, N_BUCKETS - 1)  

f_bucketCenter(int zi, int bj) =>
    float bot  = array.get(ZONE_BOTTOM, zi)
    float top  = array.get(ZONE_TOP,    zi)
    float band = (top - bot) / N_BUCKETS
    bot + band * float(bj) + band * 0.5

f_kalmanUpdate(int zi, float new_obs) =>
    float Q      = 0.005
    float R      = 0.5
    float p_prev = array.get(kalman_p,    zi)
    float p_pred = p_prev + Q
    float gain   = p_pred / (p_pred + R)
    float p_new  = (1.0 - gain) * p_pred

    array.set(kalman_p,    zi, p_new)
    array.set(kalman_gain, zi, gain)           

    float prev_gk = array.get(gravity_k, zi)
    float gk_new  = na(prev_gk) ? new_obs : prev_gk + gain * (new_obs - prev_gk)
    array.set(gravity_k, zi, gk_new)
    gk_new

f_updateGravity(float src, float delta_w) =>
    int zi = f_getZone(src)

    if zi >= 0
        int   bj    = f_getBucket(src, zi)
        int   idx   = zi * N_BUCKETS + bj

        array.set(visit_raw,   idx, array.get(visit_raw,   idx) + 1.0)
        array.set(visit_delta, idx, array.get(visit_delta, idx) + math.abs(delta_w))

        float best_w  = -1.0
        int   best_bj = 0
        for j = 0 to N_BUCKETS - 1
            float w = array.get(visit_delta, zi * N_BUCKETS + j)
            if w > best_w
                best_w  := w
                best_bj := j

        float raw_g = f_bucketCenter(zi, best_bj)
        array.set(gravity_raw, zi, raw_g)
        f_kalmanUpdate(zi, raw_g)

    zi

f_transitionQuality(int curr_z, int prev_z, float src) =>
    int result = 0
    if curr_z != prev_z and prev_z >= 0 and curr_z >= 0
        float grav  = array.get(gravity_k, prev_z)
        if not na(grav)
            float bot   = array.get(ZONE_BOTTOM, prev_z)
            float top   = array.get(ZONE_TOP,    prev_z)
            float width = top - bot
            float dist  = math.abs(src - grav)
            result := dist < width * 0.25 ? 1 : -1
    result

f_porPressure(float por_val, int zi) =>
    int state = 0
    if zi >= 0 and not na(por_val)
        float grav  = array.get(gravity_k, zi)
        if not na(grav)
            float bot       = array.get(ZONE_BOTTOM, zi)
            float top       = array.get(ZONE_TOP,    zi)
            float tolerance = (top - bot) * 0.15   
            if por_val > grav + tolerance
                state := 1
            else if por_val < grav - tolerance
                state := -1
    state

f_sdp_cum_delta() =>
    float _tw       = math.max(high, close[1]) - math.max(open, close)
    float _bw       = math.min(open, close)    - math.min(low, close[1])
    float _body     = math.abs(open - close)
    float _tw_sq    = _tw * _tw
    float _bw_sq    = _bw * _bw
    float _body_sq  = _body * _body
    float _total_sq = _tw_sq + _bw_sq + _body_sq
    bool  _bullish  = (_bw + (close - open) - _tw) >= 0
    float _buy_sh   = _bw_sq + (_bullish     ? _body_sq : 0)
    float _sell_sh  = _tw_sq + (not _bullish ? _body_sq : 0)
    float _buy_pct  = _total_sq > 0 ? _buy_sh  / _total_sq : 0.5
    float _sell_pct = _total_sq > 0 ? _sell_sh / _total_sq : 0.5
    float _net_tick = nz(volume, 0) * (_buy_pct - _sell_pct)
    ta.cum(_net_tick)

float sdp_ce_d3 = request.security(f_sdp_tk(sdp_atm - 3*sdp_step, "C"), timeframe.period, f_sdp_cum_delta(), ignore_invalid_symbol=true)
float sdp_pe_d3 = request.security(f_sdp_tk(sdp_atm - 3*sdp_step, "P"), timeframe.period, f_sdp_cum_delta(), ignore_invalid_symbol=true)
float sdp_ce_d4 = request.security(f_sdp_tk(sdp_atm - 4*sdp_step, "C"), timeframe.period, f_sdp_cum_delta(), ignore_invalid_symbol=true)
float sdp_pe_d4 = request.security(f_sdp_tk(sdp_atm - 4*sdp_step, "P"), timeframe.period, f_sdp_cum_delta(), ignore_invalid_symbol=true)
float sdp_ce_d5 = request.security(f_sdp_tk(sdp_atm - 5*sdp_step, "C"), timeframe.period, f_sdp_cum_delta(), ignore_invalid_symbol=true)
float sdp_pe_d5 = request.security(f_sdp_tk(sdp_atm - 5*sdp_step, "P"), timeframe.period, f_sdp_cum_delta(), ignore_invalid_symbol=true)

// ─── Drawing Engine ──────────────────────────────────────────────────────────
var box[]   sdp_boxes       = array.new<box>()
var label[] sdp_stat_labels = array.new<label>()
var label   sdp_debug_lbl   = na
var float[] sdp_base_net    = array.new_float(11, na)
var int     sdp_anchor_time = na

float[] sdp_net = array.from(
  nz(sdp_ce_u5) - nz(sdp_pe_u5),
  nz(sdp_ce_u4) - nz(sdp_pe_u4),
  nz(sdp_ce_u3) - nz(sdp_pe_u3),
  nz(sdp_ce_u2) - nz(sdp_pe_u2),
  nz(sdp_ce_u1) - nz(sdp_pe_u1),
  nz(sdp_ce_0)  - nz(sdp_pe_0),
  nz(sdp_ce_d1) - nz(sdp_pe_d1),
  nz(sdp_ce_d2) - nz(sdp_pe_d2),
  nz(sdp_ce_d3) - nz(sdp_pe_d3),
  nz(sdp_ce_d4) - nz(sdp_pe_d4),
  nz(sdp_ce_d5) - nz(sdp_pe_d5)
 )

bool sdp_in_session = timeframe.isintraday ? not na(time(timeframe.period, "0915-1530")) : true
bool sdp_prev_in_session = bar_index > 0 ? sdp_in_session[1] : false
bool sdp_session_open = sdp_in_session and not sdp_prev_in_session
bool sdp_new_week = ta.change(time("W")) != 0
bool sdp_new_month = ta.change(time("M")) != 0

bool sdp_reset_bar = sdp_session_open
if sdp_reset_mode == "Weekly"
    sdp_reset_bar := sdp_session_open and sdp_new_week
else if sdp_reset_mode == "Monthly"
    sdp_reset_bar := sdp_session_open and sdp_new_month

if sdp_active and (sdp_reset_bar or na(sdp_anchor_time))
    for i = 0 to 10
        array.set(sdp_base_net, i, array.get(sdp_net, i))
    sdp_anchor_time := time

if barstate.islast
    if array.size(sdp_boxes) > 0
        for i = 0 to array.size(sdp_boxes) - 1
            box.delete(array.get(sdp_boxes, i))
        array.clear(sdp_boxes)

    if array.size(sdp_stat_labels) > 0
        for i = 0 to array.size(sdp_stat_labels) - 1
            label.delete(array.get(sdp_stat_labels, i))
        array.clear(sdp_stat_labels)

    if sdp_show and sdp_active
        float sdp_max = 0.0
        for i = 0 to 10
            sdp_max := math.max(sdp_max, math.abs(array.get(sdp_net, i)))

        float min_net = array.min(sdp_net)
        float max_net = array.max(sdp_net)
        int   sdp_x1 = bar_index + sdp_offset
        float step_f = float(sdp_step)

        for i = 0 to 10
            float val        = array.get(sdp_net, i)
            float base_val   = array.get(sdp_base_net, i)
            float change_pct = f_sdp_change_pct(val, base_val)
            float strike_lvl = float(sdp_atm + ((5 - i) * sdp_step))

            float width_pct = sdp_max > 0 ? (math.abs(val) / sdp_max) : 0.0
            int   sdp_x2    = sdp_x1 + math.max(10, int(math.round(sdp_width * width_pct)))

            color clr = min_net == max_net
              ? color.rgb(128, 128, 128)
              : color.from_gradient(val, min_net, max_net, color.rgb(255, 0, 64), color.rgb(64, 255, 0))

            int alpha = math.max(10, int(math.round(75 - (65 * width_pct))))
            color bg  = color.new(clr, alpha)
            color brd = clr

            box bx = box.new(sdp_x1, strike_lvl + (step_f * 0.30),
                             sdp_x2, strike_lvl - (step_f * 0.30),
                             bgcolor=bg, border_color=brd,
                             xloc=xloc.bar_index,
                             text=str.tostring(int(strike_lvl)),
                             text_size=size.normal,
                             text_color=color.black,
                             text_halign=text.align_center,
                             text_valign=text.align_center,
                             force_overlay=true)
            array.push(sdp_boxes, bx)

            label stat_lbl = label.new(sdp_x2 + 1, strike_lvl, f_sdp_change_text(change_pct),
              xloc=xloc.bar_index, yloc=yloc.price, style=label.style_label_left,
              color=color.new(color.black, 0), textcolor=f_sdp_change_color(change_pct),
              size=size.large, force_overlay=true)
            array.push(sdp_stat_labels, stat_lbl)

        if sdp_show_diag
            int spot_atm = int(math.round(close / float(sdp_step)) * float(sdp_step))
            float atm_net = array.get(sdp_net, 5)
            float atm_base = array.get(sdp_base_net, 5)
            float atm_change = f_sdp_change_pct(atm_net, atm_base)
            string diag_txt = "═══ SDP DIAG ═══" +
             "\nRoot: " + syminfo.root + " | Step: " + str.tostring(sdp_step) +
             "\nATM input: " + str.tostring(sdp_atm) +
             "\nSpot: " + str.tostring(close) + " → Suggested ATM: " + str.tostring(spot_atm) +
             "\nReset: " + sdp_reset_mode +
             "\nATM Net: " + str.tostring(atm_net, "#.##") +
             "\nATM Base: " + str.tostring(atm_base, "#.##") +
             "\nATM Change: " + f_sdp_change_text(atm_change)
            if na(sdp_debug_lbl)
                sdp_debug_lbl := label.new(bar_index + 5, close, diag_txt,
                  xloc=xloc.bar_index, yloc=yloc.price, style=label.style_label_left,
                  color=color.new(color.black, 20), textcolor=color.white, size=size.normal,
                  force_overlay=true)
            else
                label.set_xy(sdp_debug_lbl, bar_index + 5, close)
                label.set_text(sdp_debug_lbl, diag_txt)
        else if not na(sdp_debug_lbl)
            label.delete(sdp_debug_lbl)
            sdp_debug_lbl := na
    else if not na(sdp_debug_lbl)
        label.delete(sdp_debug_lbl)
        sdp_debug_lbl := na
